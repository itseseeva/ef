
# --- Чёрные линии рамки ---------------------------------------------------
# Пиксель считается "чёрным", если ВСЕ три канала <= этого значения. У
# настоящей рамки (11, 10, 10); у тёмного леса/тени фон обычно >= 30.
FRAME_DARK_MAX = 24
# Линия рамки под полупрозрачной табличкой другого игрока — 25-40 (см. panel_detector.read_panel):
# такая "тусклая" линия годится только в паре с настоящей чёрной.
FRAME_LINE_DIM_MAX = 45
# ...и должна покрыть хотя бы такую долю длины чёрной (просветы между табличками допустимы).
FRAME_DIM_COVER_MIN = 0.85

# --- Где рамка лежит внутри зоны панели (190x90, TARGET_PANEL_* в vision_config) ---------
# Замер по кадрам 2026-10-05: верхняя линия рамки — строка 48 зоны, начинается в x=27 (как его меряет read_panel).
# Нужно поиску панели по всему экрану (panel_detector.locate_panel): игрок может перетащить
# панель куда угодно, а зону бот ставит так, чтобы рамка легла ровно сюда.
PANEL_LINE_X_IN_ROI = 27
PANEL_LINE_Y_IN_ROI = 48
# Сколько кандидатов-линий проверять за один поиск (защита от тёмных сцен: ~1 мс на кандидата).
LOCATE_MAX_CANDIDATES = 60
